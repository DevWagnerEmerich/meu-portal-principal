# ============================================
# CONFIGURAÇÃO DO MERCADOPAGO - WEBHOOK
# ============================================

# PASSO 1: Configure a URL do Webhook no painel do MercadoPago
# --------------------------------------------
# Acesse: https://www.mercadopago.com.br/developers/panel/app
# Vá em "Webhooks" ou "Notificações"
# 
# Para DESENVOLVIMENTO LOCAL:
#   1. Instale ngrok: https://ngrok.com/download
#   2. Execute: ngrok http 3001
#   3. Copie a URL gerada (ex: https://abc123.ngrok.io)
#   4. Configure no MP: https://abc123.ngrok.io/api/payment/webhook
#
# Para PRODUÇÃO:
#   Configure: https://seu-dominio.com/api/payment/webhook
#   Exemplo: https://portaleducacional.vercel.app/api/payment/webhook

# PASSO 2: Marque os eventos
# --------------------------------------------
# ✅ Pagamentos
# ✅ Envios (Mercado Pago)

# PASSO 3: Copie o Webhook Secret gerado
# --------------------------------------------
# Após salvar, o MercadoPago mostrará um "Secret"
# Copie esse valor e cole abaixo:

MERCADOPAGO_WEBHOOK_SECRET=seu_webhook_secret_aqui
